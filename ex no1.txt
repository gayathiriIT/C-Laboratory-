#include <stdio.h>
int main()
{
int a,b;

printf("arithmethic operators");
printf("%d",a+b);
printf("%d",a-b);
printf("%d",a/b);
printf("%d",a*b);
printf("%d",a%b);
printf("relational operator");
printf("%d",a==b);
printf("%d",a>=b);
printf("%d",a<=b);
printf("%d",a!=b);
printf("logical operator");
printf("%d",(a>b)&&(a<b));
printf("%d",(a>b)||(a<b));
printf("%d",!(a<b));
printf("assigment operator");

printf("%d",a-=b);
printf("%d",a*=b);
printf("%d",a/=b);
printf("%d",a%=b);
printf("increment/decrement");

}
